'use client';
import {useEffect,useRef,useState} from 'react';
import {Share2,X} from 'lucide-react';
import banner from './share-banner.json';
import {photoShareData} from './share-file';
export default function SharePage(){
 const [open,setOpen]=useState(false),[status,setStatus]=useState(''),[url,setUrl]=useState(''),[busy,setBusy]=useState(false);
 const dialog=useRef<HTMLDialogElement>(null),trigger=useRef<HTMLButtonElement>(null);
 useEffect(()=>{if(open){dialog.current?.showModal();}else if(dialog.current?.open){dialog.current.close();trigger.current?.focus();}},[open]);
 function show(){setUrl(window.location.href);setStatus('');setOpen(true);}
 async function share(){
  if(busy)return;
  setBusy(true);setStatus('');
  try{
   if(typeof navigator.share==='function'){
    await navigator.share(photoShareData(url)??{title:'Water Rules NZ',text:'Drinking water guidance. Official sources.',url});
    setStatus('Shared successfully.');
   }else{
    await navigator.clipboard.writeText(url);
    setStatus('Link copied. Paste it into your chosen app.');
   }
  }catch(e){if((e as Error).name!=='AbortError')setStatus('Sharing is unavailable in this browser. You can share the address from your browser menu.');}
  finally{setBusy(false);}
 }
 return <div className="share-page"><button ref={trigger} className="share-toggle" onClick={show} aria-label="Share this page" aria-haspopup="dialog"><Share2 size={17}/><span>Share this page</span></button><dialog ref={dialog} className="share-dialog" aria-labelledby="share-title" onCancel={()=>setOpen(false)} onClose={()=>setOpen(false)} onClick={e=>{if(e.target===dialog.current){const rect=dialog.current.getBoundingClientRect();if(e.clientX<rect.left||e.clientX>rect.right||e.clientY<rect.top||e.clientY>rect.bottom)setOpen(false);}}}>
 <div className="share-dialog-heading"><div><small>PASS IT ON</small><h2 id="share-title">Share Water Rules NZ</h2></div><button aria-label="Close sharing" onClick={()=>setOpen(false)}><X size={20}/></button></div>
 <img className="share-preview" src={banner.image} alt="Water Rules NZ — a stream through Aotearoa"/>
 <div className="share-actions"><button className="share-device" disabled={busy} onClick={share}><Share2 size={18}/>Share this page</button></div>
 {status&&<p role="status" aria-live="polite">{status}</p>}
 </dialog></div>;
}
