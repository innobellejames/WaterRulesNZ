export function createAssessment(course:LearningCourse){const questions=[...course.questions];for(let i=questions.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[questions[i],questions[j]]=[questions[j],questions[i]];}return questions.map(q=>{const choices=q.options.map((text,index)=>({text,correct:index===q.answer}));for(let i=choices.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[choices[i],choices[j]]=[choices[j],choices[i]];}return {...q,choices};});}
const progressKey='waterrules-learning-read-v2';
function LessonActivity({activity}:{activity:LearningActivity}){const [choice,setChoice]=useState<number|null>(null),[checked,setChecked]=useState(false);return <section className="lesson-activity" aria-label="Apply this lesson"><p className="eyebrow">TRY IT · NO SCORE RECORDED</p><h4>{activity.prompt}</h4><div>{activity.options.map((option,i)=><button key={option} aria-pressed={choice===i} className={choice===i?'selected':''} onClick={()=>{setChoice(i);setChecked(false);}}>{option}</button>)}</div><button className="tool-secondary" disabled={choice===null} onClick={()=>setChecked(true)}>Check your reasoning</button>{checked&&<p role="status" className="lesson-feedback"><strong>{choice===activity.answer?'Good reasoning.':'Consider this approach.'}</strong> {activity.feedback}</p>}</section>}

 useEffect(()=>{if(progressReady)try{localStorage.setItem(progressKey,JSON.stringify(progress));}catch{}},[progress,progressReady]);

 function choose(id:string){if(busy)return;setSelected(id);setLesson(0);setAssessment([]);setAnswers({});setAttempt(null);setStatus('');setFile(null);const url=new URL(location.href);url.hash=id?'learn?topic='+encodeURIComponent(id):'learn';history.replaceState(null,'',url);}
 function start(){if(!course||busy)return;window.speechSynthesis?.cancel();setSpeaking(false);setAssessment(createAssessment(course));setAnswers({});setAttempt(null);setStartedAt(new Date().toISOString());setStatus('');setFile(null);}
 function submit(){if(!course||count!==30||submitted)return;const submittedAt=new Date().toISOString();setAttempt(Object.freeze({id:crypto.randomUUID(),courseId:course.id,courseTitle:course.title,contentVersion:data.reviewed,startedAt,submittedAt,score,total:30,answered:30}));setStatus(eligible?'Threshold met. You can create an optional learning completion record.':'Review the explanations, revisit the lessons and try again.');setTimeout(()=>document.getElementById('assessment-result')?.scrollIntoView({block:'center',behavior:'smooth'}),0);}

If your browser did not save it, use the download link below.');}catch(e){setStatus((e as Error).message||'The PDF could not be created. Please try again.');}finally{setBusy(false);}}
 function listen(){if(!course)return;if(speaking){window.speechSynthesis.cancel();setSpeaking(false);return;}const l=course.lessons[lesson],utterance=new SpeechSynthesisUtterance([l.title,l.intro,...l.points,l.example].filter(Boolean).join('. '));utterance.lang='en-NZ';utterance.rate=.92;utterance.onend=()=>setSpeaking(false);utterance.onerror=()=>setSpeaking(false);window.speechSynthesis.cancel();window.speechSynthesis.speak(utterance);setSpeaking(true);}

No qualification credits, professional accreditation, verified competency or regulatory approval are awarded.</p>
 {!course?<><label className="learning-search"><Search size={18}/><span className="sr-only">Find a learning topic</span><input value={search} onChange={e=>setSearch(e.target.value)} placeholder="Find a topic — calibration, storage, source water…"/></label><p className="tool-note">{visible.length} topics · 6 lessons per topic · 30 assessment questions per topic. Lesson progress is marked by you and saved in this browser.</p><div className="learning-grid">{visible.map(c=><article key={c.id}><small>INTRODUCTORY · SIX LESSONS</small><h2>{c.title}</h2><p>{c.description}</p><p className="lesson-completion">{progress[c.id]?.length||0}/6 lessons marked read</p><button className="tool-primary" onClick={()=>choose(c.id)}>Explore topic <ArrowRight size={16}/></button></article>)}</div>{!visible.length&&<p>No matching topics. Try a broader search.</p>}</>:<>
 <button className="tool-secondary" disabled={busy} onClick={()=>choose('')}><ArrowLeft size={16}/> All topics</button><div className="learning-topic"><p className="eyebrow">INTRODUCTORY RESOURCE · CONTENT REVIEW {data.reviewed}</p><h2>{course.title}</h2><p>{course.description}</p>
 {!assessment.length?<><div className="learning-topic-tools"><span><BookOpen size={17}/> {(progress[course.id]||[]).length}/6 lessons marked read</span>{readAloud&&<button className="tool-secondary" onClick={listen}>{speaking?<Square size={16}/>:<Headphones size={16}/>} {speaking?'Stop reading':'Listen to this lesson'}</button>}<button className="tool-secondary" onClick={()=>downloadLearningNotes(course)}><Download size={16}/> Topic notes</button></div>
 <nav className="lesson-tabs" aria-label="Lessons">{course.lessons.map((l,i)=><button key={l.title} aria-current={lesson===i?'step':undefined} aria-pressed={lesson===i} onClick={()=>setLesson(i)}><span>{(progress[course.id]||[]).includes(i)?'✓':i+1}</span>{l.title}</button>)}</nav>

 <details className="lesson-reflect"><summary>Reflect or discuss</summary><p>What would you look for in your own supply or project? Describe one piece of evidence you would check and one question for an experienced colleague. Keep personal, security and sensitive cultural information out of these notes.</p><textarea aria-label="Your private lesson reflection" rows={3} value={reflection} onChange={e=>{setReflection(e.target.value);try{sessionStorage.setItem(`waterrules-learning-note:${selected}:${lesson}`,e.target.value);}catch{}}} placeholder="My observation, evidence to check, or question…"/><small>Notes stay in this browser session.  <section className="learning-assessment-intro"><h3>Check your understanding</h3><p>The 30-question quiz is optional and unproctored.  <details className="learning-sources"><summary>Read further with the original sources</summary><ul>{(course.sources||[{title:'Official supporting source',url:course.source}]).map(s=><li key={s.url}><a href={s.url} target="_blank" rel="noopener noreferrer">{s.title} ↗</a></li>)}</ul><p>These links support further reading.  <div className="assessment-progress"><strong>{submitted?`Result: ${score}/30 · ${(score/30*100).toFixed(1)}%`:`${count} of 30 answered`}</strong><progress max={30} value={count}/><span>Completion threshold: at least 26/30. Answer every question before submitting.</span></div><form onSubmit={e=>{e.preventDefault();submit();}}>{assessment.map((q,i)=><fieldset className="assessment-question" key={q.id} disabled={submitted}><legend>{i+1}. {q.question}</legend>{q.choices.map((o,j)=><label key={o.text}><input type="radio" required name={q.id} checked={answers[q.id]===j} onChange={()=>setAnswers(a=>({...a,[q.id]:j}))}/><span>{o.text}</span></label>)}{submitted&&<div className={q.choices[answers[q.id]]?.correct?'answer-correct':'answer-review'}><p><strong>{q.choices[answers[q.id]]?.correct?'Correct.':'Review.'}</strong> {q.explanation}</p><a href={q.source||course.source} target="_blank" rel="noreferrer">Read supporting source ↗</a></div>}</fieldset>)}{!submitted&&<button className="tool-primary" type="submit" disabled={count!==30}>Submit quiz</button>}</form>
