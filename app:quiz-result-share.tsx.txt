"use client";
import {useState} from "react";
import {Share2,Copy,Download,ExternalLink} from "lucide-react";
export function quizResultText(score:number,total:number,day:string,attempt:number,url:string){return `My WaterRules NZ knowledge quiz result: ${score}/${total} (${Math.round(score/total*100)}%).\n${day} · Attempt ${attempt}\nA learning quiz, not a compliance assessment.\nTry the quiz: ${url}`;}
export default function QuizResultShare({score,total,day,attempt}:{score:number;total:number;day:string;attempt:number}){
 const [status,setStatus]=useState("");const url="https://waterrulesnz.com/#quiz";
 const text=quizResultText(score,total,day,attempt,url);
 async function copy(){try{await navigator.clipboard.writeText(text);setStatus("Result copied. Paste it into your post or message.");}catch{setStatus("Select and copy the result text below.");}}
 async function share(){if(navigator.share){try{await navigator.share({title:"My WaterRules NZ quiz result",text,url});setStatus("Sharing opened.");return;}catch(e){if((e as Error).name==="AbortError")return;}}await copy();}
 function download(){const link=URL.createObjectURL(new Blob([text+"\n"],{type:"text/plain"}));const a=document.createElement("a");a.href=link;a.download=`WaterRules-quiz-result-${day}.txt`;a.click();setTimeout(()=>URL.revokeObjectURL(link),1000);}
 return <section className="quiz-result-share" aria-label="Share quiz result"><h3>Share your result</h3><div className="quiz-share-actions"><button className="tool-primary" onClick={share}><Share2 size={17}/> Share result</button><button className="tool-secondary" onClick={copy}><Copy size={16}/> Copy result</button><button className="tool-secondary" onClick={download}><Download size={16}/> Save result</button><a className="tool-secondary" href={`https://wa.me/?text=${encodeURIComponent(text)}`} target="_blank" rel="noreferrer">WhatsApp <ExternalLink size={14}/></a><a className="tool-secondary" href={`https://www.facebook.com/sharer/sharer.php?u=${encodeURIComponent(url)}`} target="_blank" rel="noreferrer" onClick={()=>void copy()}>Facebook <ExternalLink size={14}/></a><a className="tool-secondary" href={`https://www.linkedin.com/sharing/share-offsite/?url=${encodeURIComponent(url)}`} target="_blank" rel="noreferrer" onClick={()=>void copy()}>LinkedIn <ExternalLink size={14}/></a></div><p className="tool-note">Facebook and LinkedIn open the quiz page; paste your copied result into the post. The sharing menu and WhatsApp include your score.</p><textarea aria-label="Quiz result text" readOnly rows={5} value={text}/>{status&&<p role="status">{status}</p>}</section>;
}
